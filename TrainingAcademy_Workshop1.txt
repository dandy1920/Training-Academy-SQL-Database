## 1. DDL
-- ============================================================
-- 1. CREATE DATABASE
-- ============================================================

CREATE DATABASE TrainingAcademy;

USE TrainingAcademy;

-- ============================================================
-- 2. CREATE STUDENTS TABLE
-- ============================================================

CREATE TABLE Students (
    StudentID INT PRIMARY KEY,
    FullName VARCHAR(100) NOT NULL,
    Email VARCHAR(100) UNIQUE,
    BirthDate DATE
);


-- ============================================================
-- 3. CREATE INSTRUCTORS TABLE
-- ============================================================

CREATE TABLE Instructors (
    InstructorID INT PRIMARY KEY,
    FullName VARCHAR(100) NOT NULL,
    Specialization VARCHAR(100)
);


-- ============================================================
-- 4. CREATE COURSES TABLE
-- Each course belongs to one instructor.
-- ============================================================

CREATE TABLE Courses (
    CourseID INT PRIMARY KEY,
    CourseName VARCHAR(100) NOT NULL,
    Fee DECIMAL(10,2),
    InstructorID INT,
    FOREIGN KEY (InstructorID)
        REFERENCES Instructors(InstructorID)
);

-- ============================================================
-- 5. CREATE ENROLLMENTS TABLE
-- Connects students with courses.
-- ============================================================

CREATE TABLE Enrollments (
    EnrollmentID INT PRIMARY KEY,
    StudentID INT,
    CourseID INT,
    EnrollmentDate DATE,
    FOREIGN KEY (StudentID)
        REFERENCES Students(StudentID),
    FOREIGN KEY (CourseID)
        REFERENCES Courses(CourseID)
);

-- ============================================================
-- 6. CREATE SESSIONS TABLE
-- ============================================================

CREATE TABLE Sessions (
    SessionID INT PRIMARY KEY,
    CourseID INT,
    SessionDate DATE,
    Topic VARCHAR(200),
    FOREIGN KEY (CourseID)
        REFERENCES Courses(CourseID)
);

-- ============================================================
-- 7. CREATE ATTENDANCE TABLE
-- ============================================================

CREATE TABLE Attendance (
    AttendanceID INT PRIMARY KEY,
    SessionID INT,
    StudentID INT,
    Status VARCHAR(20),
    FOREIGN KEY (SessionID)
        REFERENCES Sessions(SessionID),
    FOREIGN KEY (StudentID)
        REFERENCES Students(StudentID)
);

## 3. DQL

SELECT *
FROM Students;


-- ============================================================
-- 15. SELECT - SPECIFIC COLUMNS
-- ============================================================

SELECT FullName, Email
FROM Students;


-- ============================================================
-- 16. WHERE - STUDENTS BORN AFTER 2001
-- ============================================================

SELECT *
FROM Students
WHERE BirthDate > '2001-01-01';


-- ============================================================
-- 17. WHERE - COURSES WITH FEE GREATER THAN 150
-- ============================================================

SELECT *
FROM Courses
WHERE Fee > 150;


-- ============================================================
-- 18. ORDER BY - MOST EXPENSIVE FIRST
-- ============================================================

SELECT *
FROM Courses
ORDER BY Fee DESC;


-- ============================================================
-- 19. JOIN - COURSES WITH THEIR INSTRUCTORS
-- ============================================================

SELECT
    Courses.CourseName,
    Courses.Fee,
    Instructors.FullName AS Instructor
FROM Courses
JOIN Instructors
    ON Courses.InstructorID = Instructors.InstructorID;


-- ============================================================
-- 20. JOIN - STUDENTS AND THEIR COURSES
-- ============================================================

SELECT
    Students.FullName AS Student,
    Courses.CourseName AS Course
FROM Enrollments
JOIN Students
    ON Enrollments.StudentID = Students.StudentID
JOIN Courses
    ON Enrollments.CourseID = Courses.CourseID;


-- ============================================================
-- 21. GROUP BY - STUDENTS PER COURSE
-- ============================================================

SELECT
    CourseID,
    COUNT(*) AS NumberOfStudents
FROM Enrollments
GROUP BY CourseID;


-- ============================================================
-- 22. GROUP BY + JOIN - COURSE NAMES AND STUDENT COUNTS
-- ============================================================

SELECT
    Courses.CourseName,
    COUNT(Enrollments.StudentID) AS NumberOfStudents
FROM Courses
JOIN Enrollments
    ON Courses.CourseID = Enrollments.CourseID
GROUP BY Courses.CourseName;

SELECT *

-- ============================================================
-- 24. SHOW ABSENT STUDENTS

SELECT *

-- ============================================================
-- 25. ATTENDANCE WITH STUDENT NAMES
-- ============================================================

SELECT
    Students.FullName,
    Attendance.Status
FROM Attendance

    ON Attendance.StudentID = Students.StudentID;


-- ============================================================
-- 26. COMPLETE ATTENDANCE DETAILS
-- ============================================================

SELECT
    Students.FullName AS Student,
    Sessions.SessionDate,
    Sessions.Topic,
    Attendance.Status
FROM Attendance
JOIN Students
    ON Attendance.StudentID = Students.StudentID
JOIN Sessions
    ON Attendance.SessionID = Sessions.SessionID;


-- ============================================================
-- 27. BUSINESS QUESTION:
-- How many sessions did each student attend?
-- ============================================================

SELECT
    Students.FullName,
    COUNT(*) AS PresentSessions
FROM Attendance
JOIN Students
    ON Attendance.StudentID = Students.StudentID
WHERE Attendance.Status = 'Present'
GROUP BY Students.FullName;


-- ============================================================
-- 28. ATTENDANCE SUMMARY
-- ============================================================

SELECT
    Students.FullName,
    COUNT(*) AS TotalSessions,
    SUM(
        CASE
            WHEN Attendance.Status = 'Present' THEN 1
            ELSE 0
        END
    ) AS PresentSessions
FROM Attendance
JOIN Students
    ON Attendance.StudentID = Students.StudentID
GROUP BY Students.FullName;


-- ============================================================
-- 29. ATTENDANCE PERCENTAGE
-- ============================================================

SELECT
    Students.FullName,
    COUNT(*) AS TotalSessions,
    SUM(
        CASE
            WHEN Attendance.Status = 'Present' THEN 1
            ELSE 0
        END
    ) AS PresentSessions,
    CAST(
        SUM(
            CASE
                WHEN Attendance.Status = 'Present' THEN 1
                ELSE 0
            END
        ) * 100.0 / COUNT(*)
        AS DECIMAL(5,2)
    ) AS AttendancePercentage
FROM Attendance
JOIN Students
    ON Attendance.StudentID = Students.StudentID
GROUP BY Students.FullName;
